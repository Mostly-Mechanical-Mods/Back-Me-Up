BMU_pressed_claim_update_top_contender = {
	if = {
		limit = { NOT = { exists = scope:pressed_claim }}
		$PRESSED_CLAIM$ = { save_scope_as = pressed_claim }
	}
	scope:pressed_claim = {
		every_in_list = {
			variable = contenders
			limit = { var:status = flag:contender }
			BMU_update_confidence = yes
		}
		ordered_in_list = {
			variable = contenders
			limit = { NOT = { story_owner = scope:pressed_claim.var:claim_target }}
			order_by = var:confidence_score
			scope:pressed_claim = {
				set_variable = {
					name = top_contender
					value = prev.story_owner
				}
			}
		}
	}
}

BMU_execute_bargain = {
	debug_log = "BMU_execute_bargain"
	debug_log_scopes = no
	var:pressed_claim ?= { debug_log_scopes = no }
	var:contender.story_owner ?= { debug_log_scopes = no }
	var:supporter.story_owner ?= { debug_log_scopes = no }
	var:bargain_type ?= { debug_log_scopes = no }
	save_scope_as = executed_bargain
	if = {
		limit = { var:bargain_type = flag:FF_ambition_get_council_position }
		var:council_position ?= { debug_log_scopes = no }
		switch = {
			trigger = var:council_position
			flag:councillor_chancellor = { save_scope_value_as = { name = councillor_chancellor value = yes }}
			flag:councillor_steward = { save_scope_value_as = { name = councillor_steward value = yes }}
			flag:councillor_marshal = { save_scope_value_as = { name = councillor_marshal value = yes }}
			flag:councillor_spymaster = { save_scope_value_as = { name = councillor_spymaster value = yes }}
		}
		BMU_bargain_council_position_post_effect = {
			PRESSED_CLAIM = scope:executed_bargain.var:pressed_claim
			CONTENDER = scope:executed_bargain.story_owner
			SUPPORTER = scope:executed_bargain.var:supporter.story_owner
		}
	}
	else_if = {
		limit = { var:bargain_type = flag:BMU_bargain_owned_title }
		var:title ?= { debug_log_scopes = no }
		BMU_bargain_owned_title_post_effect = {
			PRESSED_CLAIM = scope:executed_bargain.var:pressed_claim
			CONTENDER = scope:executed_bargain.story_owner
			SUPPORTER = scope:executed_bargain.var:supporter.story_owner
			TARGET = var:title
		}
	}
	clear_saved_scope = executed_bargain
}

BMU_transfer_claimed_press = {
	$PRESSED_CLAIM$ = { save_scope_as = pressed_claim }
	$CLAIMED_TITLE$ = { save_scope_as = claimed_title }
	$OLD_CLAIM_TARGET$ = { save_scope_as = old_claim_target }
	$NEW_CLAIM_TARGET$ = { save_scope_as = new_claim_target }

	if = {
		limit = { BMU_debug = yes }
		debug_log = "BMU_transfer_claimed_press"
		scope:pressed_claim = { debug_log_scopes = no }
		scope:claimed_title = { debug_log_scopes = no }
		scope:old_claim_target = { debug_log_scopes = no }
		scope:new_claim_target = { debug_log_scopes = no }
	}

	scope:old_claim_target = {
		random_pressed_claim = {
			limit = { this = scope:pressed_claim }
			save_scope_as = old_claim_target_stakeholder_story
		}
	}
	scope:new_claim_target = {
		random_contender_story = {
			limit = { var:pressed_claim_story = scope:pressed_claim }
			save_scope_as = new_claim_target_stakeholder_story
		}
	}
	
	if = {
		limit = { exists = scope:new_claim_target_stakeholder_story }
		scope:new_claim_target_stakeholder_story = {
			set_variable = {
				name = status
				value = flag:contender
			}
			set_variable = {
				name = contender
				value = scope:new_claim_target_stakeholder_story
			}
			# every_in_list = {
			# 	variable = BMU_stakeholder_ambitions
			# 	if = {
			# 		limit = { has_variable = bargained_with }
			# 		var:bargained_with = {
			# 			remove_list_variable = {
			# 				name = bargains
			# 				target = scope:new_claim_target_stakeholder_story
			# 			}
			# 		}
			# 	}
			# 	end_story = yes
			# }
		}
	}
	else = {
		scope:old_claim_target_stakeholder_story = {
			make_story_owner = scope:new_claim_target
			set_variable = {
				name = stakeholder
				value = scope:new_claim_target
			}
		}
	}
	scope:pressed_claim = {
		set_variable = {
			name = claim_target
			value = scope:new_claim_target
		}
		make_story_owner = scope:new_claim_target
		every_in_list = {
			variable = claimants
			limit = {
				NOR = {
					has_claim_on = scope:claimed_title
					any_heir_title = { this = scope:claimed_title }
				}
			}
			prev = {
				remove_list_variable = {
					name = claimants
					target = prev
				}
			}
		}
	}
	scope:new_claim_target = {
		every_heir = {
			scope:pressed_claim = {
				add_to_variable_list = {
					name = claimants
					target = prev
				}
			}
		}
	}
}

BMU_dropout_supporters_of = {
	$PRESSED_CLAIM$ = { save_temporary_scope_as = pressed_claim_temp }

	scope:pressed_claim_temp = {
		every_in_list = {
			variable = supporters
			limit = { var:contender.story_owner = scope:contender_temp }
			if = {
				limit = { BMU_debug = yes }
				set_variable = { name = end_story value = flag:BMU_dropout_supporters_of }
			}
			end_story = yes
		}
	}

	BMU_update_pressed_claim_gui = { PRESSED_CLAIM = $PRESSED_CLAIM$ }
}

BMU_dropout_contender = {
	$CONTENDER$ = {
		every_contender_story = {
			limit = { var:pressed_claim_story = $PRESSED_CLAIM$ }
			end_story = yes
		}
	}

	if = {
		limit = {
			$PRESSED_CLAIM$ = {
				any_in_list = {
					variable = contenders
					NOR = {
						story_owner = var:pressed_claim_story.story_owner
						story_owner = $CONTENDER$
					}
					count = 0
				}
			}
		}
		custom_tooltip = {
			text = BMU_PRESSED_CLAIM_NO_CONTENDERS
			$PRESSED_CLAIM$ = { end_story = yes }
		}
	}

	BMU_update_pressed_claim_gui = { PRESSED_CLAIM = $PRESSED_CLAIM$ }
}

BMU_create_bargain = {
	hidden_effect = {
		$PRESSED_CLAIM$ = { save_scope_as = pressed_claim_temp }
		$CONTENDER$ = {
			save_scope_as = contender_char
			random_contender_story = {
				limit = { var:pressed_claim_story = $PRESSED_CLAIM$ }
				save_scope_as = contender_temp
			}
		}
		$SUPPORTER$ = {
			save_scope_as = supporter_char
			random_supporter_story = {
				limit = {
					story_type = BMU_supporter_story
					var:pressed_claim_story = $PRESSED_CLAIM$
				}
				save_scope_as = supporter_temp
			}
			if = {
				limit = { NOT = { exists = scope:supporter_temp }}
				random_stakeholder_story = {
					limit = { var:pressed_claim_story = $PRESSED_CLAIM$ }
					save_scope_as = supporter_temp
				}
			}
			else = { custom_tooltip = BMU_BUG }
		}
		if = {
			limit = { BMU_debug = yes }
			debug_log = "BMU_create_bargain"
			scope:pressed_claim_temp = { debug_log_scopes = no }
			scope:contender_char = { debug_log_scopes = no }
			scope:contender_temp = { debug_log_scopes = no }
			scope:supporter_char = { debug_log_scopes = no }
			scope:supporter_temp = { debug_log_scopes = no }
		}

		scope:contender_char = {
			create_story = {
				type = BMU_bargain_story
				save_scope_as = bargain
			}
			if = {
				limit = { exists = scope:bargain }
				scope:bargain = {
					set_variable = {
						name = pressed_claim
						value = scope:pressed_claim_temp
					}
					set_variable = {
						name = contender
						value = scope:contender_temp
					}
					set_variable = {
						name = supporter
						value = scope:supporter_temp
					}
					
					if = {
						limit = { exists = scope:FF_ambition_get_council_position }
						if = { limit = { BMU_debug = yes } debug_log = "FF_ambition_get_council_position" }
						if = { limit = { BMU_bargain_council_position_setup_trigger_global = yes }}
						set_variable = { name = bargain_type value = flag:FF_ambition_get_council_position }
						if = { limit = { scope:councillor_chancellor = yes } set_variable = { name = council_position value = flag:councillor_chancellor }}
						else_if = { limit = { scope:councillor_steward = yes } set_variable = { name = council_position value = flag:councillor_steward }}
						else_if = { limit = { scope:councillor_marshal = yes } set_variable = { name = council_position value = flag:councillor_marshal }}
						else_if = { limit = { scope:councillor_spymaster = yes } set_variable = { name = council_position value = flag:councillor_spymaster }}
						else_if = { limit = { scope:councillor_court_chaplain = yes } set_variable = { name = council_position value = flag:councillor_court_chaplain }}
						else_if = { limit = { scope:councillor_castellan = yes } set_variable = { name = council_position value = flag:councillor_castellan }}
						else_if = { limit = { scope:councillor_admiral = yes } set_variable = { name = council_position value = flag:councillor_admiral }}
						set_variable = { name = willingness_cost value = scope:contender_char.BMU_willingness_bargain_council_position }
						set_variable = { name = support_score value = scope:supporter_temp.story_owner.BMU_ambition_get_council_position }
						if = { limit = { BMU_debug = yes } var:council_position = { debug_log_scopes = no }}
					}
					else_if = {
						limit = { exists = scope:BMU_bargain_owned_title exists = scope:target }
						if = { limit = { BMU_debug = yes } debug_log = "BMU_bargain_owned_title" }
						set_variable = { name = bargain_type value = flag:BMU_bargain_owned_title }
						set_variable = { name = title value = scope:target }
						set_variable = { name = willingness_cost value = scope:contender_char.BMU_willingness_bargain_title }
						set_variable = { name = support_score value = scope:supporter_temp.story_owner.BMU_ambition_get_title }
						if = { limit = { BMU_debug = yes } var:title = { debug_log_scopes = no }}
					}
					else = { if = { limit = { BMU_debug = yes } debug_log = "no ambition" }}
				}
				scope:contender_temp = {
					add_to_variable_list = {
						name = bargains
						target = scope:bargain
					}
				}
				scope:supporter_temp = {
					set_variable = {
						name = bargain
						value = scope:bargain
					}
				}
			}
		}
		
		scope:supporter_temp ?= {
			BMU_calc_best_contenders = { PRESSED_CLAIM = scope:pressed_claim_temp }
		}
	
		clear_saved_scope = pressed_claim_temp
		clear_saved_scope = contender_temp
		clear_saved_scope = supporter_temp

	}
	
}

	# Jealousy check
	if = {
		limit = {
			any_in_list = {
				list = titles_to_grant
				any_claimant = { grant_title_rivalry_trigger = yes }
			}
		}
		hidden_effect = {
			random = {
				chance = 25
				ordered_in_list = {
					list = titles_to_grant
					limit = {
						any_claimant = { grant_title_rivalry_trigger = yes }
					}
					order_by = tier
					save_scope_as = claim_title
					ordered_claimant = {
						limit = { grant_title_rivalry_trigger = yes }
						order_by = primary_title.tier
						save_scope_as = claim_vassal
						if = {
							limit = {
								$SUPPORTER$ = { is_lowborn = yes }
							}
							$SUPPORTER$ = { save_scope_as = lowborn_scope }
						}
						else_if = {
							limit = {
								$SUPPORTER$ = { has_claim_on = scope:claim_title }
							}
							$SUPPORTER$ = { save_scope_as = claimant_scope }
						}
						$CONTENDER$ = {
							trigger_event = {
								id = bp1_yearly.8070
								days = 5
							}
						}
					}
				}
			}
		}
	}

	#Stress impact
	$CONTENDER$ = {
		if = {
			limit = {
				any_in_list = {
					list = target_titles_bargained
					trigger_if = {
						limit = { tier = tier_barony }
						title_province.barony = {
							has_wrong_holding_type = no
						}
					}
					count >= 1
				}
			}
			
			# Generous characters enjoy giving away titles
			if = {
				limit = {
					has_trait = generous
				}
				add_stress = generous_title_grant_value
			}

			# Gnostics lose stress for giving away titles.
			if = {
				limit = {
					$CONTENDER$ = {
						faith = {
							has_doctrine_parameter = granting_titles_gives_stress
						}
					}
				}
				$CONTENDER$ = {
					add_stress = {
						value = 0

						# Base stress value scaled on title tier.
						every_in_list = {
							list = target_titles_bargained

							add = {
								value = 0

								if = {
									limit = { this.tier = tier_county }
									add = minor_stress_loss # 10 base
								}
								else_if = {
									limit = { this.tier = tier_duchy }
									add = {
										value = medium_stress_loss
										multiply = 1.25 # 25 base
									}
								}
								else_if = {
									limit = { this.tier = tier_kingdom }
									add = {
										value = major_stress_loss
										multiply = 2 # 80 base
									}
								}
								else_if = {
									limit = { this.tier = tier_empire }
									add = {
										value = monumental_stress_loss 
										multiply = 2 # 200 base
									}
								}

								# Lose more stress for giving away titles of your primary tier; lose less stress for giving away titles far below your primary in tier.
								multiply = {
									value = this.tier
									add = 1
									divide = $CONTENDER$.primary_title.tier
								}

								# Round to the nearest multiple of 5.
								divide = 5
								round = yes
								multiply = 5

								# For a Duke, this will be x1.25 for Duchies (-30) and x1 for Counties(-10)
								# For a King, this will be x1.25 for a Kingdom (-100), x1 for a Duchy (-25) and x0.75 for Counties(-10)
								# For an Emperor, this will be x1.25 for an Empire(-200), x1 for a Kingdom(-80), x0.75 for a Duchy(-20) and x0.4 for a Counties (-5)
							}
						}
					}
				}
			}

			# Greedy & ambitious ruler gain stress for giving away titles if they would normally be able to keep them.
			if = {
				limit = {
					# Pre-filter 
					OR = {
						has_trait = greedy
						has_trait = ambitious
					}
					domain_size <= domain_limit # Not over my domain limit
					any_in_list = {
						list = target_titles_bargained
						tier <= tier_county # At least 1 selected title counts toward domain limit
					}

					# Special exemption for lieges who are over their held duchy limit, which are giving away that duchy (and associated counties) specifically
					trigger_if = {
						limit = {
							has_too_many_held_duchies_trigger = yes
						}
						NOT = {				
							# Granting at least 1 duchy
							any_in_list = {
								list = target_titles_bargained
								title_counts_towards_too_many_duchies_trigger = yes
								save_temporary_scope_as = granted_duchy

								# All counties being granted are De Jure part of the duchy that is being granted
								any_in_list = {
									list = target_titles_bargained
									count = all 
									trigger_if = {
										limit = { tier = tier_county}
										de_jure_liege = scope:granted_duchy
									}
								}
							}
						}
					}
				}
				# Actual stress gain effects
				if = {
					limit = {
						domain_size = domain_limit
					}
					stress_impact = {
						greedy = minor_stress_impact_gain
						ambitious = minor_stress_impact_gain
					}
				}
				else_if = {
					limit = {
						domain_size < domain_limit
					}
					stress_impact = {
						greedy = medium_stress_impact_gain
						ambitious = medium_stress_impact_gain
					}
				}
			}
		}

		# Maintenance for runestones
		if = {
			limit = {
				# Use expanded version if we have FP1.
				has_fp1_dlc_trigger = yes
				fp1_remove_stele_new_holder_trigger = {
					TITLE = scope:target
					PREVIOUS_HOLDER = $CONTENDER$
					NEW_HOLDER = $SUPPORTER$
				}
			}
			# Actually removed in fp1_other_decisions.0113
			custom_tooltip = runestone_grant_title_warning
		}
		else_if = {
			limit = {
				scope:target = {
					exists = var:ancestor_to_bury
					has_county_modifier = county_raised_runestone_modifier
				}
				NOT = { $SUPPORTER$.dynasty = var:ancestor_to_bury.dynasty } #We only have to change something if the new holder is of a different dynasty
			}
			# Actually removed in religious_decision.0312
			custom_tooltip = runestone_grant_title_warning
		}
		
		if = {
			limit = {
				exists = $CONTENDER$.dynasty
				$CONTENDER$.dynasty = {
					has_dynasty_perk = ep1_culture_legacy_3
				}
				$SUPPORTER$ = {
					is_landed = no
					NOT = {
						culture = $CONTENDER$.culture
					}
					any_in_list = {
						list = target_titles_bargained
						OR = {
							tier = tier_county
							tier = tier_barony
						}
						culture = $SUPPORTER$.culture
					}
				}
			}
			$CONTENDER$ = {
				if = {
					limit = {
						can_add_hook = {
							target = $SUPPORTER$
							type = favor_hook
						}
					}
					add_hook = {
						type = favor_hook
						target = $SUPPORTER$
					}
				}
			}
		}

		#Feedback for actor
		hidden_effect = {
			if = {
				limit = {
					any_in_list = {
						list = target_titles_bargained
						count = 1
						NOT = { tier = $CONTENDER$.highest_held_title_tier }
					}
					NOT = { exists = scope:recipient_is_independent }
					$SUPPORTER$ = {
						any_held_title = {
							tier > tier_barony
							NOT = {
								is_in_list = target_titles_bargained
							}
						}
					}
				}
				random_in_list = {
					list = target_titles_bargained
					save_temporary_scope_as = loc_title
				}
				send_interface_toast = {
					title = grant_titles_interaction_notification
				
					left_icon = $CONTENDER$					
					right_icon = $SUPPORTER$
				
					custom_tooltip = grant_titles_interaction_notification_effect_2
					if = {
						limit = {
							exists = dynasty
							is_diff_faith_or_culture_trigger = {
								CHAR = $SUPPORTER$
								STATUS = involved
							}
							any_character_struggle = {
								involvement = involved
								is_struggle_type = iberian_struggle
								has_struggle_phase_parameter = struggle_grant_titles_diff_faith_culture_gives_prestige

							}
						}
						dynasty = {
							add_dynasty_prestige = minor_dynasty_prestige_gain
						}

					}
				}
			}
			else_if = {
				limit = {
					any_in_list = {
						list = target_titles_bargained
						count > 1
						NOT = { tier = $CONTENDER$.highest_held_title_tier }
					}
					NOT = { exists = scope:recipient_is_independent }
					$SUPPORTER$ = {
						any_held_title = {
							tier > tier_barony
							NOT = {
								is_in_list = target_titles_bargained
							}
						}
					}
				}
				random_in_list = {
					list = target_titles_bargained
					save_temporary_scope_as = loc_title
				}
				send_interface_toast = {
					title = grant_titles_interaction_notification
				
					left_icon = $CONTENDER$					
					right_icon = $SUPPORTER$
				
					custom_tooltip = grant_titles_interaction_notification_effect_3
				}
			}
			else_if = {
				limit = {
					exists = scope:recipient_is_independent
				}

				random_in_list = {
					list = target_titles_bargained
					save_temporary_scope_as = loc_title
				}
				send_interface_toast = {
					title = grant_titles_interaction_notification
				
					left_icon = $CONTENDER$					
					right_icon = $SUPPORTER$
				
					custom_tooltip = grant_titles_interaction_notification_effect_6
				}
			}
			else_if = {
				limit = {
					$SUPPORTER$ = {
						any_in_list = { #Became independent from you, but still have the same liege
							list = target_titles_bargained
							tier = $CONTENDER$.highest_held_title_tier
						}
						top_liege = $CONTENDER$.top_liege
					}
				}
				send_interface_toast = {
					title = grant_titles_interaction_notification
				
					left_icon = $CONTENDER$					
					right_icon = $SUPPORTER$
				
					custom_tooltip = grant_titles_interaction_notification_effect_5
				}
			}
			else_if = {
				limit = {
					$SUPPORTER$ = {
						any_in_list = { #Became independent
							list = target_titles_bargained
							tier = $CONTENDER$.highest_held_title_tier
						}
					}
				}
				send_interface_toast = {
					title = grant_titles_interaction_notification
				
					left_icon = $CONTENDER$					
					right_icon = $SUPPORTER$
				
					custom_tooltip = grant_titles_interaction_notification_effect_4
				}
			}
			else = {
				send_interface_toast = {
					title = grant_titles_interaction_notification

					left_icon = $CONTENDER$					
					right_icon = $SUPPORTER$

					custom_tooltip = grant_titles_interaction_notification_effect
				}
			}
		}
	}

	hidden_effect = {
		# Struggle Catalyst
		if = {
			limit = {
				$CONTENDER$ = {
					is_diff_faith_or_culture_trigger = {
						CHAR = $SUPPORTER$
						STATUS = involved
					}
					any_character_struggle = {
						involvement = involved
						activate_struggle_catalyst_secondary_character_involvement_involved_trigger = {
							CATALYST = catalyst_grant_privilege_to_diff_faith_culture_vassal
							CHAR = $SUPPORTER$
						}
					}
				}
			}
			$CONTENDER$ = {
				every_character_struggle = {
					involvement = involved
					limit = {
						activate_struggle_catalyst_secondary_character_involvement_involved_trigger = {
							CATALYST = catalyst_grant_privilege_to_diff_faith_culture_vassal
							CHAR = $SUPPORTER$
						}
					}
					activate_struggle_catalyst = {
						catalyst = catalyst_grant_privilege_to_diff_faith_culture_vassal
						character = $CONTENDER$
					}
				}
			}
		}
	}
