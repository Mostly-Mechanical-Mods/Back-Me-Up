BMU_bargain_story = {
	on_setup = {
	}
	
	on_end = {
		var:contender = {
			remove_list_variable = {
				name = bargains
				target = prev
			}
		}
		var:supporter = {
			remove_variable = bargain
		}
	}

	on_owner_death = {
	}

	effect_group = {
		weeks = 1
		triggered_effect = {
			trigger = { BMU_debug = yes }
			effect = {
				debug_log = "BMU_bargain_story weekly"
				debug_log_scopes = yes
				story_owner = { debug_log_scopes = no }
				if = {
					limit = { has_variable = bargain_type }
					var:bargain_type = { debug_log_scopes = no }
				}
				else = { debug_log = "missing ambition"}
			}
		}
		triggered_effect = {
			trigger = {
				NAND = {
					switch = {
						trigger = var:bargain_type
						flag:FF_ambition_get_council_position = {
							switch = {
								trigger = var:council_position
								flag:councillor_chancellor = {
									BMU_bargain_council_position_is_valid = { PRESSED_CLAIM = scope:story.var:pressed_claim CONTENDER = scope:story.var:contender.story_owner SUPPORTER = scope:story.var:supporter.story_owner COUNCIL_POSITION = councillor_chancellor }
								}
								flag:councillor_steward = {
									BMU_bargain_council_position_is_valid = { PRESSED_CLAIM = scope:story.var:pressed_claim CONTENDER = scope:story.var:contender.story_owner SUPPORTER = scope:story.var:supporter.story_owner COUNCIL_POSITION = councillor_steward }
								}
								flag:councillor_marshal = {
									BMU_bargain_council_position_is_valid = { PRESSED_CLAIM = scope:story.var:pressed_claim CONTENDER = scope:story.var:contender.story_owner SUPPORTER = scope:story.var:supporter.story_owner COUNCIL_POSITION = councillor_marshal }
								}
								flag:councillor_spymaster = {
									BMU_bargain_council_position_is_valid = { PRESSED_CLAIM = scope:story.var:pressed_claim CONTENDER = scope:story.var:contender.story_owner SUPPORTER = scope:story.var:supporter.story_owner COUNCIL_POSITION = councillor_spymaster }
								}
								flag:councillor_court_chaplain = {
									BMU_bargain_council_position_is_valid = { PRESSED_CLAIM = scope:story.var:pressed_claim CONTENDER = scope:story.var:contender.story_owner SUPPORTER = scope:story.var:supporter.story_owner COUNCIL_POSITION = councillor_court_chaplain }
								}
								flag:councillor_castellan = {
									BMU_bargain_council_position_is_valid = { PRESSED_CLAIM = scope:story.var:pressed_claim CONTENDER = scope:story.var:contender.story_owner SUPPORTER = scope:story.var:supporter.story_owner COUNCIL_POSITION = councillor_castellan }
								}
								flag:councillor_admiral = {
									BMU_bargain_council_position_is_valid = { PRESSED_CLAIM = scope:story.var:pressed_claim CONTENDER = scope:story.var:contender.story_owner SUPPORTER = scope:story.var:supporter.story_owner COUNCIL_POSITION = councillor_admiral }
								}
							}
						}
						flag:BMU_bargain_owned_title = {
							BMU_can_bargain_titles = {
								PRESSED_CLAIM = scope:story.var:pressed_claim
								CONTENDER = scope:story.var:contender.story_owner
								SUPPORTER = scope:story.var:supporter.story_owner
							}
							BMU_title_can_be_bargained = {
								GIVER = scope:story.var:contender.story_owner
								TAKER = scope:story.var:supporter.story_owner
								TITLE = scope:story.var:title
							}
						}
					}
				}
			}
			effect = {
				if = {
					limit = { BMU_debug = yes }
					debug_log = "BMU_bargain_story weekly Bargain invalid"
					debug_log = "Claimed Title: [THIS.Var('pressed_claim').Var('claimed_title').Title.GetNameNoTooltip]"
					debug_log = "Contender: [THIS.Var('contender').Var('stakeholder').Char.GetNameNoTooltip]"
					debug_log = "Supporter: [THIS.Var('supporter').Var('stakeholder').Char.GetNameNoTooltip]"
					debug_log = "Bargain type: [THIS.Var('bargain_type').GetFlagName]"
					var:council_position ?= { prev = { debug_log = "Council Position: [THIS.Var('council_position').GetFlagName]" }}
					debug_log = "Willingness Cost: [THIS.Var('willingness_cost').GetValue|0]"
					debug_log = "Gambit Cost: [THIS.Var('gambit_cost').GetValue|0]"
					switch = {
						trigger = var:bargain_type
						flag:FF_ambition_get_council_position = {
							scope:story.var:supporter.story_owner = {
								if = {
									limit = { is_capable_adult = no }
									scope:story = { set_variable = { name = invalidate_reason value = flag:BMU_BARGAIN_INVALIDATED_SUPPORTER_NOT_CAPABLE_ADULT }}
								}
								else_if = {
									limit = { is_imprisoned = yes }
									scope:story = { set_variable = { name = invalidate_reason value = flag:BMU_BARGAIN_INVALIDATED_SUPPORTER_IMPRISONED }}
								}
								else_if = {
									limit = { is_at_war_with = scope:story.story_owner }
									scope:story = { set_variable = { name = invalidate_reason value = flag:BMU_BARGAIN_INVALIDATED_SUPPORTER_AT_WAR_WITH_CONTENDER }}
								}
								else_if = {
									limit = {
										has_variable = block_hire_councillor
										var:block_hire_councillor = scope:story.story_owner
									}
									scope:story = { set_variable = { name = invalidate_reason value = flag:BMU_BARGAIN_INVALIDATED_SUPPPORTER_BLOCKED_FROM_CONTENDER_COUNCIL }}
								}
								else_if = {
									limit = {
										NOT = { liege = scope:story.story_owner }
										NOT = { BMU_is_claim_target_vassal_that_follows_pressed_claim = { PRESSED_CLAIM = scope:story.var:pressed_claim }}
									}
									scope:story = { set_variable = { name = invalidate_reason value = flag:BMU_BARGAIN_INVALIDATED_SUPPPORTER_NOT_VASSAL_OF_PRESSED_CLAIM_WINNER }}
								}
								else = {
									if = {
										limit = {
											var:council_position = flag:councillor_chancellor
											NOT = { BMU_can_be_councillor_chancellor_of = { COURT_OWNER = scope:story.story_owner }}
										}
										scope:story = { set_variable = { name = invalidate_reason value = flag:BMU_BARGAIN_INVALIDATED_SUPPPORTER_CANNOT_BE_CHANCELLOR }}
									}
									else_if = {
										limit = {
											var:council_position = flag:councillor_steward
											NOT = { BMU_can_be_councillor_steward_of = { COURT_OWNER = scope:story.story_owner }}
										}
										scope:story = { set_variable = { name = invalidate_reason value = flag:BMU_BARGAIN_INVALIDATED_SUPPPORTER_CANNOT_BE_STEWARD }}
									}
									else_if = {
										limit = {
											var:council_position = flag:councillor_marshal
											NOT = { BMU_can_be_councillor_marshal_of = { COURT_OWNER = scope:story.story_owner }}
										}
										scope:story = { set_variable = { name = invalidate_reason value = flag:BMU_BARGAIN_INVALIDATED_SUPPPORTER_CANNOT_BE_MARSHAL }}
									}
									else_if = {
										limit = {
											var:council_position = flag:councillor_spymaster
											NOT = { BMU_can_be_councillor_spymaster_of = { COURT_OWNER = scope:story.story_owner }}
										}
										scope:story = { set_variable = { name = invalidate_reason value = flag:BMU_BARGAIN_INVALIDATED_SUPPPORTER_CANNOT_BE_SPYMASTER }}
									}
									else_if = {
										limit = {
											var:council_position = flag:councillor_court_chaplain
											NOT = { BMU_can_be_councillor_court_chaplain_of = { COURT_OWNER = scope:story.story_owner }}
										}
										scope:story = { set_variable = { name = invalidate_reason value = flag:BMU_BARGAIN_INVALIDATED_SUPPPORTER_CANNOT_BE_COURT_CHAPLAIN }}
									}
								}
							}
						}
					}
				}
				story_owner = {
					send_interface_toast = {
						title = BMU_BARGAIN_INVALIDATED
						left_icon = scope:story.var:supporter.var:stakeholder
						custom_tooltip = BMU_DEBUG
					}
				}
				scope:story.var:supporter.story_owner = {
					send_interface_toast = {
						title = BMU_BARGAIN_INVALIDATED
						left_icon = scope:story.story_owner
						custom_tooltip = BMU_DEBUG
					}
				}
				end_story = yes
			}
		}
	}
}
